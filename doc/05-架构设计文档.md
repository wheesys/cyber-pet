pub struct WindowManager {

impl WindowManager {
    pub fn create_pet_window(&mut self, pet_id: &str) -> Result<Window> {
        let window = WindowBuilder::new()
            .title("Pet")
            .transparent(true)
            .decorations(false)
            .always_on_top(true)
            .build()?;
        
        self.windows.insert(pet_id.to_string(), window);

            window.set_position(Position::Physical(PhysicalPosition { x, y }));
        }
    }
}

#### 3.3.3 P2P网络模块

**职责**：管理libp2p连接、节点发现、数据传输

**连接策略**：先连中继，后台升级直连（参考《04-技术选型决策》第2节）

**Rust实现**：
```rust
// src-tauri/src/p2p/manager.rs
pub struct P2PManager {
    swarm: Swarm<Behaviour>,
    relay_nodes: Vec<Multiaddr>,
}

impl P2PManager {
    pub async fn connect_to_peer(&mut self, peer_id: PeerId) -> Result<()> {
        // 1. 立即通过中继连接
        let relay_addr = self.select_best_relay();
        self.swarm.dial(relay_addr.with(Protocol::P2pCircuit))?;
        
        // 2. 后台尝试直连升级
        tokio::spawn(async move {
            if let Ok(direct_conn) = try_direct_connection(peer_id).await {
                upgrade_connection(direct_conn);
            }
        });
        
        Ok(())
    }
}
```

#### 3.3.4 AI服务模块

**职责**：管理AI调用、路由决策、缓存

**前端实现**：
```typescript
// src/services/ai.service.ts
class AIService {
    private schedulerAI: AIClient;
    private simpleAI: AIClient;
    private complexAI: AIClient;
    
    async chat(message: string): Promise<string> {
        // 1. 调度AI判断复杂度
        const complexity = await this.schedulerAI.judgeComplexity(message);
        
        // 2. 路由到对应AI
        const ai = complexity === 'simple' ? this.simpleAI : this.complexAI;
        
        // 3. 检查缓存
        const cached = await this.getCache(message);
        if (cached) return cached;
        
        // 4. 调用AI
        const response = await ai.chat(message);
        
        // 5. 缓存结果
        await this.setCache(message, response);
        
        return response;
    }
}
```

            transparent: true,
            antialias: true,

    }

### 5.2 P2P核心实现

```rust
// cyber-pet-core/src/p2p/mod.rs
pub struct P2PCore {
    swarm: Swarm<Behaviour>,
    relay_nodes: Vec<Multiaddr>,
}

impl P2PCore {
    pub async fn connect(&mut self, peer_id: PeerId) -> Result<ConnectionId> {
        // 立即连接中继
        let relay = self.select_relay();
        let conn_id = self.dial_relay(relay, peer_id).await?;
        
        // 后台升级直连
        self.spawn_direct_upgrade(peer_id);
        
        Ok(conn_id)
    }
    
    pub async fn send(&mut self, peer_id: PeerId, data: Vec<u8>) -> Result<()> {
        self.swarm.behaviour_mut().send_message(peer_id, data)
    }
}
```

### 5.3 FFI绑定

#### Android JNI绑定

```rust
// cyber-pet-core/src/ffi/android.rs
#[no_mangle]
pub extern "C" fn Java_com_cyberpet_p2p_P2PManager_nativeConnect(
    env: JNIEnv,
    _class: JClass,
    peer_id: JString,
) -> jboolean {
    let peer_id: String = env.get_string(peer_id).unwrap().into();
    // 调用P2PCore
    match P2P_CORE.lock().unwrap().connect(peer_id) {
        Ok(_) => JNI_TRUE,
        Err(_) => JNI_FALSE,
    }
}
```

// 使用AES-256加密
import CryptoJS from 'crypto-js';

function encryptAPIKey(key: string): string {
  const deviceKey = getDeviceKey(); // 基于硬件信息生成
  return CryptoJS.AES.encrypt(key, deviceKey).toString();
}

function decryptAPIKey(encrypted: string): string {
  const deviceKey = getDeviceKey();
  const bytes = CryptoJS.AES.decrypt(encrypted, deviceKey);
  return bytes.toString(CryptoJS.enc.Utf8);
}
